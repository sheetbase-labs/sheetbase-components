    const value = text(scalarOf(args[0]));
    const oldText = text(scalarOf(args[1]));
    const newText = text(scalarOf(args[2]));
    if (oldText === "") return value;
    const instance = args[3] === undefined ? undefined : Math.trunc(number(scalarOf(args[3])));
    if (instance === undefined) return value.split(oldText).join(newText);
    let index = -1;
    for (let seen = 0; seen < instance; seen++) {
      index = value.indexOf(oldText, index + 1);
      if (index < 0) return value;
    }
    return value.slice(0, index) + newText + value.slice(index + oldText.length);
  },

  // --- date / time ---
  TIME: (args) =>
    dayFraction(
      args[0] === undefined ? 0 : number(scalarOf(args[0])),
      args[1] === undefined ? 0 : number(scalarOf(args[1])),
      args[2] === undefined ? 0 : number(scalarOf(args[2])),
    ),
  HOUR: (args) => asDate(scalarOf(args[0])).getUTCHours(),
  MINUTE: (args) => asDate(scalarOf(args[0])).getUTCMinutes(),
  SECOND: (args) => asDate(scalarOf(args[0])).getUTCSeconds(),
  DATEVALUE: (args) => asDate(scalarOf(args[0])),
  TIMEVALUE: (args) => {
    const parsed = asDate(scalarOf(args[0]));
    return dayFraction(parsed.getUTCHours(), parsed.getUTCMinutes(), parsed.getUTCSeconds());
  },
  WEEKDAY: (args) => {
    const day = asDate(scalarOf(args[0])).getUTCDay();
    const type = args[1] === undefined ? 1 : Math.trunc(number(scalarOf(args[1])));
    if (type === 1) return day + 1;
    if (type === 2) return day === 0 ? 7 : day;
    if (type === 3) return day === 0 ? 6 : day - 1;
    return day + 1;
  },
  WEEKNUM: (args) => weekNumber(asDate(scalarOf(args[0])), args[1] === undefined ? 1 : Math.trunc(number(scalarOf(args[1])))),
  ISOWEEKNUM: (args) => {
    const dateValue = asDate(scalarOf(args[0]));
    const target = new Date(Date.UTC(dateValue.getUTCFullYear(), dateValue.getUTCMonth(), dateValue.getUTCDate()));
    const dayNumber = (target.getUTCDay() + 6) % 7;
    target.setUTCDate(target.getUTCDate() - dayNumber + 3);
    const firstThursday = new Date(Date.UTC(target.getUTCFullYear(), 0, 4));
    const firstDayNumber = (firstThursday.getUTCDay() + 6) % 7;
    firstThursday.setUTCDate(firstThursday.getUTCDate() - firstDayNumber + 3);
    return 1 + Math.round((target.getTime() - firstThursday.getTime()) / (7 * 86_400_000));
  },
  EDATE: (args) => {
    const dateValue = asDate(scalarOf(args[0]));
    const months = Math.trunc(number(scalarOf(args[1])));
    const shifted = new Date(dateValue);
    shifted.setUTCDate(1);
    shifted.setUTCMonth(shifted.getUTCMonth() + months);
    const lastDay = new Date(Date.UTC(shifted.getUTCFullYear(), shifted.getUTCMonth() + 1, 0)).getUTCDate();
    shifted.setUTCDate(Math.min(dateValue.getUTCDate(), lastDay));
    return shifted;
  },
  EOMONTH: (args) => {
    const dateValue = asDate(scalarOf(args[0]));
    const months = Math.trunc(number(scalarOf(args[1])));
    const shifted = new Date(dateValue);
    shifted.setUTCDate(1);
    shifted.setUTCMonth(shifted.getUTCMonth() + months + 1);
    shifted.setUTCDate(0);
    return shifted;
  },
  DATEDIF: (args) => {
    const start = asDate(scalarOf(args[0]));
    const end = asDate(scalarOf(args[1]));
    const unit = text(scalarOf(args[2])).toUpperCase();
    if (end < start) throw new Error("#NUM!");
    const years = end.getUTCFullYear() - start.getUTCFullYear();
    const months = years * 12 + (end.getUTCMonth() - start.getUTCMonth());
    const days = Math.round((end.getTime() - start.getTime()) / 86_400_000);
    if (unit === "Y") return months < 12 ? 0 : end < new Date(Date.UTC(end.getUTCFullYear(), start.getUTCMonth(), start.getUTCDate())) ? years - 1 : years;
    if (unit === "M") return end < new Date(Date.UTC(end.getUTCFullYear(), start.getUTCMonth(), start.getUTCDate())) ? months - 1 : months;
    if (unit === "D") return days;
    if (unit === "YM") {
      const diff = end.getUTCMonth() - start.getUTCMonth() + (end.getUTCDate() < start.getUTCDate() ? -1 : 0);
      return (diff + 12) % 12;
    }
    if (unit === "YD") {
      const anchor = new Date(Date.UTC(end.getUTCFullYear(), start.getUTCMonth(), start.getUTCDate()));
      return Math.round((end.getTime() - (end < anchor ? anchor.getTime() - 365 * 86_400_000 : anchor.getTime())) / 86_400_000);
    }
    if (unit === "MD") return end.getUTCDate() >= start.getUTCDate() ? end.getUTCDate() - start.getUTCDate() : end.getUTCDate();
    throw new Error("#VALUE!");
  },
  NETWORKDAYS: (args, ctx) => {
    const start = asDate(scalarOf(args[0]));
    const end = asDate(scalarOf(args[1]));
    const holidays = args[2] === undefined ? [] : flatten(args[2], ctx).map(asDate);
    const holidayKeys = new Set(holidays.map((date) => date.toISOString().slice(0, 10)));
    const direction = end < start ? -1 : 1;
    let count = 0;
    const cursor = new Date(start);
    while (direction === 1 ? cursor <= end : cursor >= end) {
      const day = cursor.getUTCDay();
      const key = cursor.toISOString().slice(0, 10);
      if (day !== 0 && day !== 6 && !holidayKeys.has(key)) count += direction;
      cursor.setUTCDate(cursor.getUTCDate() + direction);
    }
    return count;
  },
  WORKDAY: (args, ctx) => {
    const start = asDate(scalarOf(args[0]));
    const days = Math.trunc(number(scalarOf(args[1])));
    const holidays = args[2] === undefined ? [] : flatten(args[2], ctx).map(asDate);
    const holidayKeys = new Set(holidays.map((date) => date.toISOString().slice(0, 10)));
    const direction = days < 0 ? -1 : 1;
    let remaining = Math.abs(days);
    const cursor = new Date(start);
    while (remaining > 0) {
      cursor.setUTCDate(cursor.getUTCDate() + direction);
      const day = cursor.getUTCDay();
      const key = cursor.toISOString().slice(0, 10);
      if (day !== 0 && day !== 6 && !holidayKeys.has(key)) remaining -= 1;
    }
    return cursor;
  },

  // --- criteria aggregation ---
  SUMIF: (args, ctx) => conditionalSum(args, ctx, "sum"),
  AVERAGEIF: (args, ctx) => conditionalSum(args, ctx, "average"),
  COUNTIF: (args, ctx) => conditionalCount(args, ctx)[0],
  SUMIFS: (args, ctx) => conditionalSumIfs(args, ctx, "sum"),
  AVERAGEIFS: (args, ctx) => conditionalSumIfs(args, ctx, "average"),
  COUNTIFS: (args, ctx) => conditionalCountIfs(args, ctx),

  // --- lookup ---
  VLOOKUP: (args, ctx) => {
    const lookup = scalarOf(args[0]);
    const rows = matrixOf(args[1], ctx);
    const columnIndex = Math.trunc(number(scalarOf(args[2])));
    const index = rows.findIndex((row) => (row[0] ?? null) === lookup);
    return index >= 0 ? rows[index]?.[columnIndex - 1] ?? null : "#N/A";
  },
  HLOOKUP: (args, ctx) => {
    const lookup = scalarOf(args[0]);
    const rows = matrixOf(args[1], ctx);
    const rowIndex = Math.trunc(number(scalarOf(args[2])));
    const header = rows[0] ?? [];
    const index = header.findIndex((value) => value === lookup);
    return index >= 0 ? rows[rowIndex - 1]?.[index] ?? null : "#N/A";
  },
  INDEX: (args, ctx) => {
    const rows = matrixOf(args[0], ctx);
    const row = args[1] === undefined ? undefined : Math.trunc(number(scalarOf(args[1])));
    const column = args[2] === undefined ? undefined : Math.trunc(number(scalarOf(args[2])));
    if (row === undefined) throw new Error("#VALUE!");
    if (column === undefined && rows.length === 1) return rows[0]?.[row - 1] ?? null;
    if (column === undefined && (rows[0]?.length ?? 0) === 1) return rows[row - 1]?.[0] ?? null;
    if (column === undefined) throw new Error("#REF!");
    return rows[row - 1]?.[column - 1] ?? null;
  },
  CHOOSE: (args) => {
    const index = Math.trunc(number(scalarOf(args[0])));
    const chosen = args[index];
    return chosen === undefined ? "#VALUE!" : scalarOf(chosen);
  },
  ROWS: (args, ctx) => matrixOf(args[0], ctx).length,
  COLUMNS: (args, ctx) => matrixOf(args[0], ctx)[0]?.length ?? 0,
  ADDRESS: (args) => {
    const row = Math.trunc(number(scalarOf(args[0])));
    const column = Math.trunc(number(scalarOf(args[1])));
    const absolute = args[2] === undefined ? 1 : Math.trunc(number(scalarOf(args[2])));
    if (row < 1 || column < 1) throw new Error("#VALUE!");
    const rowLock = absolute === 1 || absolute === 2 ? "$" : "";
    const columnLock = absolute === 1 || absolute === 3 ? "$" : "";
    return `${columnLock}${columnName(column)}${rowLock}${row}`;
  },
  TRANSPOSE: (args, ctx) => {
    const rows = matrixOf(args[0], ctx);
    const transposed: Scalar[][] = [];
    for (const row of rows) {
      row.forEach((value, index) => {
        transposed[index] ??= [];
        transposed[index].push(value);
      });
    }
    return { kind: "spill", values: transposed };
  },
  FILTER: (args, ctx) => {
    const rows = matrixOf(args[0], ctx);
    const include = matrixOf(args[1], ctx);
    const flags = include.flat();
    const filtered = rows.filter((_, index) => boolean(flags[index] ?? false));
    if (filtered.length === 0) {
      if (args[2] !== undefined) return scalarOf(args[2]);
      throw new Error("#CALC!");
    }
    return { kind: "spill", values: filtered };
  },
};

/** Every function name in the registry, sorted. Demos and docs consume this so coverage stays honest. */
export const FUNCTION_NAMES: readonly string[] = Object.keys(FUNCTIONS).sort();

/** Functions that must recalculate on every pass; the scheduler consumes this set. */
export const VOLATILE_FUNCTIONS: ReadonlySet<string> = new Set(["NOW", "TODAY", "RAND", "RANDBETWEEN"]);

export function isVolatileFunction(name: string): boolean {
  return VOLATILE_FUNCTIONS.has(name.toUpperCase());
}

function weekNumber(dateValue: Date, type: number): number {
  const startDay = type === 2 ? 1 : 0;
  const date = new Date(Date.UTC(dateValue.getUTCFullYear(), dateValue.getUTCMonth(), dateValue.getUTCDate()));
  const offset = (date.getUTCDay() - startDay + 7) % 7;
  const weekStart = new Date(date);
  weekStart.setUTCDate(date.getUTCDate() - offset);
  const yearStart = new Date(Date.UTC(weekStart.getUTCFullYear(), 0, 1));
  const yearOffset = (yearStart.getUTCDay() - startDay + 7) % 7;
  yearStart.setUTCDate(yearStart.getUTCDate() - yearOffset);
  return 1 + Math.round((weekStart.getTime() - yearStart.getTime()) / (7 * 86_400_000));
}

function confidenceNormal(args: readonly FunctionArg[]): number {
  const alpha = number(scalarOf(args[0]));
  const sd = number(scalarOf(args[1]));
  const size = Math.trunc(number(scalarOf(args[2])));
  if (alpha <= 0 || alpha >= 1 || sd <= 0 || size < 1) throw new Error("#NUM!");
  return normalInverse(1 - alpha / 2) * (sd / Math.sqrt(size));
}

function twoTailTInverse(probability: number, degreesOfFreedom: number): number {
  if (degreesOfFreedom <= 0) throw new Error("#NUM!");
  if (probability <= 0 || probability > 1) throw new Error("#NUM!");
  return invertIncreasing((x) => tCdf(x, degreesOfFreedom), 1 - probability / 2);
}

function leftTailTInverse(probability: number, degreesOfFreedom: number): number {
  if (degreesOfFreedom <= 0) throw new Error("#NUM!");
  if (probability <= 0 || probability >= 1) throw new Error("#NUM!");
  if (probability === 0.5) return 0;
  if (probability > 0.5) return invertIncreasing((x) => tCdf(x, degreesOfFreedom), probability);
  return -invertIncreasing((x) => tCdf(x, degreesOfFreedom), 1 - probability);
}

function throwNumeric(): never {
  throw new Error("#NUM!");
}

function columnName(column: number): string {
  let name = "";
  let remaining = column;
  while (remaining > 0) {
    const offset = (remaining - 1) % 26;
    name = String.fromCharCode(65 + offset) + name;
    remaining = Math.floor((remaining - 1) / 26);
  }
  return name;
}

function skewness(args: readonly FunctionArg[], ctx: FunctionContext, population: boolean): number {
  const values = trimNumeric(args.flatMap((argument) => flatten(argument, ctx)));
  const count = values.length;
  if (count < (population ? 2 : 3)) throw new Error("#DIV/0!");
  const mean = values.reduce((total, value) => total + value, 0) / count;
  const denominator = population
    ? Math.sqrt(values.reduce((total, value) => total + (value - mean) ** 2, 0) / count)
    : Math.sqrt(values.reduce((total, value) => total + (value - mean) ** 2, 0) / (count - 1));
  if (denominator === 0) throw new Error("#DIV/0!");
  const cubed = values.reduce((total, value) => total + ((value - mean) / denominator) ** 3, 0);
  return population ? cubed / count : (count / ((count - 1) * (count - 2))) * cubed;
}

function sample(args: readonly FunctionArg[], ctx: FunctionContext): number[] {
  const values = trimNumeric(args.flatMap((argument) => flatten(argument, ctx)));
  if (values.length < 2) throw new Error("#DIV/0!");
  return values;
}

/** Excel A-suffix semantics: text counts as 0, TRUE/FALSE as 1/0, numeric
 * strings as their numbers — nothing is skipped. Empty cells are skipped. */
function sampleA(args: readonly FunctionArg[], ctx: FunctionContext): number[] {
  const values = args
    .flatMap((argument) => flatten(argument, ctx))
    .filter((value) => value !== null && value !== "")
    .map((value) => {
      if (typeof value === "boolean") return value ? 1 : 0;
      if (value === null) return 0;
      const parsed = Number(value);
      return Number.isFinite(parsed) ? parsed : 0;
    });
  if (values.length < 1) throw new Error("#DIV/0!");
  return values;
}

function matrixPair(args: readonly FunctionArg[], ctx: FunctionContext): Array<[number, number]> {
  const left = flatten(args[0], ctx);
  const right = flatten(args[1], ctx);
  if (left.length !== right.length || left.length === 0) throw new Error("#N/A");
  const pairs: Array<[number, number]> = [];
  for (let index = 0; index < left.length; index++) {
    const a = left[index];
    const b = right[index];
    if (typeof a === "number" && typeof b === "number") pairs.push([a, b]);
  }
  return pairs;
}

function covariance(pairs: Array<[number, number]>, population: boolean): number {
  if (pairs.length < (population ? 1 : 2)) throw new Error("#DIV/0!");
  const meanLeft = pairs.reduce((total, [a]) => total + a, 0) / pairs.length;
  const meanRight = pairs.reduce((total, [, b]) => total + b, 0) / pairs.length;
  return (
    pairs.reduce((total, [a, b]) => total + (a - meanLeft) * (b - meanRight), 0) /
    (population ? pairs.length : pairs.length - 1)
  );
}

function correlation(pairs: Array<[number, number]>, unused: boolean): number {
  void unused;
  if (pairs.length < 2) throw new Error("#DIV/0!");
  const meanLeft = pairs.reduce((total, [a]) => total + a, 0) / pairs.length;
  const meanRight = pairs.reduce((total, [, b]) => total + b, 0) / pairs.length;
  let numerator = 0;
  let leftSquared = 0;
  let rightSquared = 0;
  for (const [a, b] of pairs) {
    numerator += (a - meanLeft) * (b - meanRight);
    leftSquared += (a - meanLeft) ** 2;
    rightSquared += (b - meanRight) ** 2;
  }
  const denominator = Math.sqrt(leftSquared * rightSquared);
  if (denominator === 0) throw new Error("#DIV/0!");
  return numerator / denominator;
}

function percentile(args: readonly FunctionArg[], ctx: FunctionContext, inclusive: boolean, quarter?: number): number {
  const values = trimNumeric(flatten(args[0], ctx)).sort((left, right) => left - right);
  if (values.length === 0) throw new Error("#NUM!");
  const fraction = quarter ?? number(scalarOf(args[1]));
  if (fraction < 0 || fraction > 1) throw new Error("#NUM!");
  const position = inclusive ? fraction * (values.length - 1) : fraction * (values.length + 1) - 1;
  const lower = Math.floor(position);
  const upper = Math.ceil(position);
  if (lower < 0 || upper >= values.length) throw new Error("#NUM!");
  const weight = position - lower;
  return values[lower] * (1 - weight) + values[upper] * weight;
}

function conditionalSum(args: readonly FunctionArg[], ctx: FunctionContext, mode: "sum" | "average"): number {
  const criteriaRange = flatten(args[0], ctx);
  const criteria = scalarOf(args[1] ?? null);
  const sumRange = args[2] === undefined ? criteriaRange : flatten(args[2], ctx);
  let total = 0;
  let count = 0;
  criteriaRange.forEach((value, index) => {
    if (matchesCriteria(value, criteria)) {
      const target = sumRange[index];
      if (typeof target === "number") {
        total += target;
        count += 1;
      }
    }
  });
  if (mode === "average") {
    if (count === 0) throw new Error("#DIV/0!");
    return total / count;
  }
  return total;
}

function conditionalCount(args: readonly FunctionArg[], ctx: FunctionContext): [number] {
  const criteriaRange = flatten(args[0], ctx);
  const criteria = scalarOf(args[1] ?? null);
  return [criteriaRange.filter((value) => matchesCriteria(value, criteria)).length];
}

function conditionalPairs(args: readonly FunctionArg[], _ctx: FunctionContext): Array<[FunctionArg, Scalar]> {
  const pairs: Array<[FunctionArg, Scalar]> = [];
  for (let index = 1; index + 1 < args.length; index += 2) {
    pairs.push([args[index], scalarOf(args[index + 1])]);
  }
  return pairs;
}

function conditionalSumIfs(args: readonly FunctionArg[], ctx: FunctionContext, mode: "sum" | "average"): number {
  const sumRange = flatten(args[0], ctx);
  const pairs = conditionalPairs(args, ctx);
  const ranges = pairs.map(([range]) => flatten(range, ctx));
  const criteriaList = pairs.map(([, criteria]) => criteria);
  let total = 0;
  let count = 0;
  for (let index = 0; index < sumRange.length; index++) {
    if (ranges.every((range, pairIndex) => matchesCriteria(range[index] ?? null, criteriaList[pairIndex]))) {
      const target = sumRange[index];
      if (typeof target === "number") {
        total += target;
        count += 1;
      }
    }
  }
  if (mode === "average") {
    if (count === 0) throw new Error("#DIV/0!");
    return total / count;
  }
  return total;
}

function conditionalCountIfs(args: readonly FunctionArg[], ctx: FunctionContext): number {
  const pairs: Array<[FunctionArg, Scalar]> = [];
  for (let index = 0; index + 1 < args.length; index += 2) {
    pairs.push([args[index], scalarOf(args[index + 1])]);
  }
  const ranges = pairs.map(([range]) => flatten(range, ctx));
  const criteriaList = pairs.map(([, criteria]) => criteria);
  let count = 0;
  for (let index = 0; index < (ranges[0]?.length ?? 0); index++) {
    if (ranges.every((range, pairIndex) => matchesCriteria(range[index] ?? null, criteriaList[pairIndex]))) count += 1;
  }
  return count;
}

/** Dispatch a function by name. Unknown names raise Excel's #NAME?. */
export function callFunction(
  name: string,
  args: readonly FunctionArg[],
  ctx: FunctionContext,
): Scalar | SpillMatrix {
  const implementation = FUNCTIONS[name.toUpperCase()];
  if (!implementation) throw new Error("#NAME?");
  return implementation(args, ctx);
}
